→ POST para backend com transação assinada

↓

→ Controller recebe requisição

→ Valida JWT

→ Service chama Web3jBroadcastService

↓

→ Web3jBroadcastService faz eth_sendRawTransaction

→ Retorna transaction receipt

↓

→ Executa setVerified(address, true)

→ Minera o bloco

→ Estado atualizado!

📄 blockchain/src/MiniIdentityRegistry.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract MiniIdentityRegistry {
    
    // Mapeamento: endereço => verificado?
    mapping(address => bool) private verified;
    
    // Evento emitido quando um endereço é verificado/não-verificado
    event VerifiedStatusUpdated(address indexed user, bool status);
    
    /**
     * @notice Define o status de verificação de um endereço
     * @param user O endereço a ser marcado
     * @param status true = verificado, false = não verificado
     */
    function setVerified(address user, bool status) external {
        verified[user] = status;
        emit VerifiedStatusUpdated(user, status);
    }
    
    /**
     * @notice Verifica se um endereço está verificado
     * @param user O endereço a verificar
     * @return true se verificado, false caso contrário
     */
    function isVerified(address user) external view returns (bool) {
        return verified[user];
    }
}

Passo 3.2: Web3jBroadcastService

📄 lib-java/src/main/java/com/miniregistry/Web3jBroadcastService.java
package com.miniregistry;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.web3j.protocol.Web3j;
import org.web3j.protocol.core.DefaultBlockParameterName;
import org.web3j.protocol.core.methods.response.EthSendTransaction;
import org.web3j.protocol.core.methods.response.TransactionReceipt;
import org.web3j.protocol.http.HttpService;

import java.io.IOException;
import java.util.Optional;

public class Web3jBroadcastService {

    private static final Logger log = LoggerFactory.getLogger(Web3jBroadcastService.class);
    private final Web3j web3j;

    public Web3jBroadcastService(String nodeUrl) {
        this.web3j = Web3j.build(new HttpService(nodeUrl));
        log.info("Web3j conectado em: {}", nodeUrl);
    }

    /**
     * Faz broadcast de uma transação assinada
     * @param signedTransaction Transação em formato hex (0x...)
     * @return Hash da transação
     */
    public String broadcastTransaction(String signedTransaction) throws IOException {
        log.info("Broadcasting transação...");
        
        EthSendTransaction response = web3j.ethSendRawTransaction(signedTransaction).send();
        
        if (response.hasError()) {
            String errorMsg = response.getError().getMessage();
            log.error("Erro no broadcast: {}", errorMsg);
            throw new RuntimeException("Erro ao fazer broadcast: " + errorMsg);
        }
        
        String txHash = response.getTransactionHash();
        log.info("Transação enviada! Hash: {}", txHash);
        
        return txHash;
    }

    /**
     * Aguarda o receipt da transação (com retry)
     * @param txHash Hash da transação
     * @return TransactionReceipt
     */
    public TransactionReceipt waitForReceipt(String txHash) throws Exception {
        log.info("Aguardando confirmação da transação: {}", txHash);
        
        int maxAttempts = 30;
        int attempt = 0;
        
        while (attempt < maxAttempts) {
            Optional receipt = web3j
                .ethGetTransactionReceipt(txHash)
                .send()
                .getTransactionReceipt();
            
            if (receipt.isPresent()) {
                log.info("✅ Transação confirmada no bloco: {}", 
                    receipt.get().getBlockNumber());
                return receipt.get();
            }
            
            attempt++;
            log.debug("Tentativa {}/{} - Aguardando confirmação...", attempt, maxAttempts);
            Thread.sleep(1000); // Aguarda 1 segundo
        }
        
        throw new RuntimeException("Timeout aguardando confirmação da transação");
    }
}

Passo 3.3: IdentityRegistryReadService

📄 lib-java/src/main/java/com/miniregistry/IdentityRegistryReadService.java
package com.miniregistry;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.web3j.abi.FunctionEncoder;
import org.web3j.abi.FunctionReturnDecoder;
import org.web3j.abi.TypeReference;
import org.web3j.abi.datatypes.Address;
import org.web3j.abi.datatypes.Bool;
import org.web3j.abi.datatypes.Function;
import org.web3j.abi.datatypes.Type;
import org.web3j.protocol.Web3j;
import org.web3j.protocol.core.DefaultBlockParameterName;
import org.web3j.protocol.core.methods.request.Transaction;
import org.web3j.protocol.http.HttpService;

import java.util.Arrays;
import java.util.List;

public class IdentityRegistryReadService {

    private static final Logger log = LoggerFactory.getLogger(IdentityRegistryReadService.class);
    private final Web3j web3j;

    public IdentityRegistryReadService(String nodeUrl) {
        this.web3j = Web3j.build(new HttpService(nodeUrl));
    }

    /**
     * Verifica se um endereço está verificado
     * @param contractAddress Endereço do contrato
     * @param userAddress Endereço do usuário
     * @return true se verificado, false caso contrário
     */
    public boolean isVerified(String contractAddress, String userAddress) throws Exception {
        log.info("Verificando status de: {}", userAddress);
        
        // 1. Criar a função ABI
        Function function = new Function(
            "isVerified",
            Arrays.asList(new Address(userAddress)),
            Arrays.asList(new TypeReference() {})
        );
        
        // 2. Encodar a função
        String encodedFunction = FunctionEncoder.encode(function);
        
        // 3. Fazer eth_call (não custa gas!)
        String result = web3j.ethCall(
            Transaction.createEthCallTransaction(null, contractAddress, encodedFunction),
            DefaultBlockParameterName.LATEST
        ).send().getValue();
        
        // 4. Decodificar o resultado
        List decoded = FunctionReturnDecoder.decode(result, function.getOutputParameters());
        
        boolean isVerified = (Boolean) decoded.get(0).getValue();
        log.info("Status de {}: {}", userAddress, isVerified ? "VERIFICADO" : "NÃO VERIFICADO");
        
        return isVerified;
    }
}

Passo 4.4: Service

📄 backend/src/main/java/com/miniregistry/service/RegistryService.java
package com.miniregistry.service;

import com.miniregistry.IdentityRegistryReadService;
import com.miniregistry.Web3jBroadcastService;
import com.miniregistry.dto.BroadcastResponse;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;
import org.web3j.protocol.core.methods.response.TransactionReceipt;

@Service
public class RegistryService {

    private static final Logger log = LoggerFactory.getLogger(RegistryService.class);

    private final Web3jBroadcastService broadcastService;
    private final IdentityRegistryReadService readService;

    @Value("${blockchain.contract.address}")
    private String contractAddress;

    public RegistryService(
        Web3jBroadcastService broadcastService,
        IdentityRegistryReadService readService
    ) {
        this.broadcastService = broadcastService;
        this.readService = readService;
    }

    public BroadcastResponse broadcast(String signedTransaction) {
        try {
            log.info("Iniciando broadcast...");
            
            // 1. Fazer broadcast
            String txHash = broadcastService.broadcastTransaction(signedTransaction);
            
            // 2. Aguardar confirmação
            TransactionReceipt receipt = broadcastService.waitForReceipt(txHash);
            
            // 3. Verificar status
            String status = receipt.isStatusOK() ? "SUCCESS" : "FAILED";
            
            log.info("✅ Broadcast concluído! Status: {}", status);
            
            return new BroadcastResponse(
                txHash,
                receipt.getBlockNumber().toString(),
                status
            );
            
        } catch (Exception e) {
            log.error("❌ Erro no broadcast", e);
            throw new RuntimeException("Erro ao processar transação: " + e.getMessage());
        }
    }

    public boolean isVerified(String address) {
        try {
            return readService.isVerified(contractAddress, address);
        } catch (Exception e) {
            log.error("Erro ao verificar endereço", e);
            throw new RuntimeException("Erro ao verificar: " + e.getMessage());
        }
    }
}

package com.miniregistry.controller;

import com.miniregistry.dto.BroadcastRequest;
import com.miniregistry.dto.BroadcastResponse;

import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

@RestController
@RequestMapping("/api/registry")
@CrossOrigin(origins = "http://localhost:3000")
public class RegistryController {

    public RegistryController(RegistryService service) {
        this.service = service;
    }

    @PostMapping("/broadcast")
    public ResponseEntity broadcast(@RequestBody BroadcastRequest request) {

    }

    @GetMapping("/is-verified")
    public ResponseEntity isVerified(@RequestParam String address) {
        boolean verified = service.isVerified(address);
        return ResponseEntity.ok(verified);
    }
}

Passo 5.3: Utilitário de Assinatura

📄 frontend/src/lib/prepareAndSignTransaction.ts
import { ethers } from "ethers";

export async function prepareAndSignTransaction({
  contractAddress,
  abi,
  functionName,
  args,
  signer,
}: {
  contractAddress: string;
  abi: any[];
  functionName: string;
  args: any[];
  signer: ethers.Signer;
}) {
  console.log("🔐 Preparando transação...");

  // 1. Criar instância do contrato
  const contract = new ethers.Contract(contractAddress, abi, signer);

  // 2. Encodar os dados da função
  const data = contract.interface.encodeFunctionData(functionName, args);

  // 3. Obter provider e configurações
  const provider = signer.provider!;
  const from = await signer.getAddress();
  const nonce = await provider.getTransactionCount(from);
  const network = await provider.getNetwork();

  // 4. Estimar gas
  const gasLimit = await contract[functionName].estimateGas(...args);

  // 5. Montar objeto da transação
  const tx: ethers.TransactionRequest = {
    to: contractAddress,
    data,
    gasLimit,
    nonce,
    chainId: network.chainId,
  };

  // 6. Configurar gas price
  const feeData = await provider.getFeeData();
  tx.type = 0; // Legacy transaction
  tx.gasPrice = feeData.gasPrice || undefined;

  console.log("📝 Transação preparada:", tx);

  // 7. ASSINAR a transação
  try {
    const signedTransaction = await signer.signTransaction(tx);
    const transactionHash = ethers.keccak256(signedTransaction);

    console.log("✅ Transação assinada!");
    console.log("Hash:", transactionHash);

    return {
      signedTransaction,
      transactionHash,
    };
  } catch (error: any) {
    // Fallback para carteiras que não suportam signTransaction
    const message = String(error?.message || "");
    if (message.includes("eth_signTransaction")) {
      console.warn("⚠️ Carteira não suporta signTransaction, usando sendTransaction");
      const response = await signer.sendTransaction(tx);
      return {
        signedTransaction: null,
        transactionHash: response.hash,
      };
    }
    throw error;
  }
}

📄 frontend/src/store/slices/registrySlice.ts
import { createSlice, createAsyncThunk } from '@reduxjs/toolkit';
import { ethers } from 'ethers';
import { prepareAndSignTransaction } from '@/lib/prepareAndSignTransaction';
import MiniRegistryABI from '@/abi/MiniIdentityRegistry.json';

const CONTRACT_ADDRESS = '0x5FbDB2315678afecb367f032d93F642f64180aa3';
const BACKEND_URL = 'http://localhost:8080/api/registry';

interface RegistryState {
  loading: boolean;
  error: string | null;
  lastTxHash: string | null;
  verifiedStatus: boolean | null;
}

const initialState: RegistryState = {
  loading: false,
  error: null,
  lastTxHash: null,
  verifiedStatus: null,
};

// THUNK 1: Broadcast de setVerified
export const broadcastSetVerified = createAsyncThunk(
  'registry/broadcastSetVerified',
  async ({ address, verified }: { address: string; verified: boolean }) => {
    // 1. Conectar MetaMask
    if (!window.ethereum) {
      throw new Error('MetaMask não encontrado!');
    }

    const provider = new ethers.BrowserProvider(window.ethereum);
    await provider.send('eth_requestAccounts', []);
    const signer = await provider.getSigner();

    console.log('👛 Conectado com:', await signer.getAddress());

    // 2. Preparar e assinar transação
    const { signedTransaction, transactionHash } = await prepareAndSignTransaction({
      contractAddress: CONTRACT_ADDRESS,
      abi: MiniRegistryABI,
      functionName: 'setVerified',
      args: [address, verified],
      signer,
    });

    // 3. Enviar para backend
    const response = await fetch(`${BACKEND_URL}/broadcast`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        signedTransaction: signedTransaction || transactionHash,
        address,
        verified,
      }),
    });

    if (!response.ok) {
      throw new Error('Erro no backend');
    }

    const result = await response.json();
    console.log('✅ Resposta do backend:', result);

    return result.transactionHash;
  }
);

// THUNK 2: Verificar status
export const checkVerified = createAsyncThunk(
  'registry/checkVerified',
  async (address: string) => {
    const response = await fetch(`${BACKEND_URL}/is-verified?address=${address}`);
    
    if (!response.ok) {
      throw new Error('Erro ao verificar');
    }

    const isVerified = await response.json();
    return isVerified;
  }
);

const registrySlice = createSlice({
  name: 'registry',
  initialState,
  reducers: {
    clearError: (state) => {
      state.error = null;
    },
  },
  extraReducers: (builder) => {
    // broadcastSetVerified
    builder
      .addCase(broadcastSetVerified.pending, (state) => {
        state.loading = true;
        state.error = null;
      })
      .addCase(broadcastSetVerified.fulfilled, (state, action) => {
        state.loading = false;
        state.lastTxHash = action.payload;
      })
      .addCase(broadcastSetVerified.rejected, (state, action) => {
        state.loading = false;
        state.error = action.error.message || 'Erro desconhecido';
      });

    // checkVerified
    builder
      .addCase(checkVerified.pending, (state) => {
        state.loading = true;
        state.error = null;
      })
      .addCase(checkVerified.fulfilled, (state, action) => {
        state.loading = false;
        state.verifiedStatus = action.payload;
      })
      .addCase(checkVerified.rejected, (state, action) => {
        state.loading = false;
        state.error = action.error.message || 'Erro desconhecido';
      });
  },
});

export const { clearError } = registrySlice.actions;
export default registrySlice.reducer;
